pipeline {
    agent any
    environment {
        PYTHON = 'C:\\Users\\Chandana\\AppData\\Local\\Programs\\Python\\Python310\\python.exe'
    }
    stages {
        stage('Run Selenium Tests with pytest') {
            steps {
                echo "Running Selenium Tests using pytest"

                // Install dependencies using Python 3.10
                bat '"%PYTHON%" -m pip install -r requirements.txt'

                // Start Flask app
                bat 'start "FlaskApp" /B "%PYTHON%" app.py'

                // Wait for Flask to start
                bat 'ping 127.0.0.1 -n 6 > nul'

                // Run the Selenium tests
                bat '"%PYTHON%" -m pytest tests/test_registrationapp.py -v'
            }
        }


        stage('Build Docker Image') {
            steps {
                echo "Build Docker Image"
                bat "docker build -t seleniumdemoapp:v1 ."
            }
        }
        stage('Docker Login') {
            steps {
                  bat 'docker login -u mudavathchandana -p chandana@123@'
                }
            }
        stage('push Docker Image to Docker Hub') {
            steps {
                echo "push Docker Image to Docker Hub"
                bat "docker tag seleniumdemoapp:v1 mudavathchandana/sample2026:seleniumtestimage"               
                    
                bat "docker push mudavathchandana/sample2026:seleniumtestimage"
                
            }
        }
        stage('Deploy to Kubernetes') { 
            steps { 
                    // apply deployment & service 
                    bat 'kubectl apply -f deployment.yaml --validate=false' 
                    bat 'kubectl apply -f service.yaml' 
            } 
        }
    }
    post {
        success {
            echo 'Pipeline completed successfully!'
        }
        failure {
            echo 'Pipeline failed. Please check the logs.'
        }
    }
}
